function bytesToBase64(bytes) {
  if (typeof bytes.toBase64 === "function") return bytes.toBase64();
  let binary = "";
  const chunk = 4096;
  for (let i = 0; i < bytes.length; i += chunk) {
    binary += String.fromCharCode.apply(null, bytes.subarray(i, i + chunk));
  }
  return btoa(binary);
}

function base64ToBytes(b64) {
  if (typeof Uint8Array["fromBase64"] === "function") return Uint8Array["fromBase64"](b64);
  const binary = atob(b64);
  const out = new Uint8Array(binary.length);
  for (let i = 0; i < binary.length; i++) out[i] = binary.charCodeAt(i);
  return out;
}

function safeSend(socket, data) {
  try {
    if (socket.readyState === 1) socket.send(data);
  } catch (_) {}
}

function safeClose(socket, code, reason) {
  try {
    const validCode = code >= 1000 && code <= 4999 && code !== 1005 && code !== 1006 ? code : 1000;
    socket.close(validCode, reason || "");
  } catch (_) {}
}

function normalizeCityKey(str) {
  return String(str || "")
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")
    .replace(/đ/gi, "d")
    .toLowerCase()
    .replace(/\b(thanh pho|tp\.?|tinh)\b/g, " ")
    .replace(/[^a-z0-9\s]/g, " ")
    .replace(/\s+/g, " ")
    .trim();
}

function describeWeatherCode(code) {
  if (code === 0) return "trời quang";
  if (code <= 3) return "ít mây đến nhiều mây";
  if (code === 45 || code === 48) return "có sương mù";
  if (code >= 51 && code <= 57) return "mưa phùn nhẹ";
  if (code >= 61 && code <= 67) return "có mưa";
  if (code >= 71 && code <= 77) return "có tuyết";
  if (code >= 80 && code <= 82) return "mưa rào";
  if (code >= 95) return "mưa dông";
  return "thời tiết thay đổi";
}

function formatVnDateTime(dateInput) {
  const d = dateInput instanceof Date ? dateInput : new Date(dateInput);
  if (Number.isNaN(d.getTime())) return String(dateInput || "");
  const parts = new Intl.DateTimeFormat("vi-VN", {
    timeZone: "Asia/Ho_Chi_Minh",
    day: "2-digit",
    month: "2-digit",
    year: "numeric",
    hour: "2-digit",
    minute: "2-digit",
    hour12: false,
  }).formatToParts(d);
  const map = {};
  for (const p of parts) map[p.type] = p.value;
  return `ngày ${map.day}/${map.month}/${map.year} lúc ${map.hour}:${map.minute}`;
}

function decodeHtml(str) {
  return String(str || "")
    .replace(/<[^>]+>/g, " ")
    .replace(/&amp;/g, "&")
    .replace(/&quot;/g, '"')
    .replace(/&#39;|&apos;/g, "'")
    .replace(/&lt;/g, "<")
    .replace(/&gt;/g, ">")
    .replace(/&nbsp;/g, " ")
    .replace(/\s+/g, " ")
    .trim();
}

async function fetchWeather(location) {
  const rawLoc = String(location || "Hà Nội").trim() || "Hà Nội";
  const key = normalizeCityKey(rawLoc);
  let target = VN_CITIES[key];

  if (!target) {
    try {
      const geoUrl = `https://geocoding-api.open-meteo.com/v1/search?name=${encodeURIComponent(rawLoc)}&count=1&language=vi&format=json`;
      const geoRes = await fetch(geoUrl, { signal: AbortSignal.timeout(2500), cf: { cacheTtl: 86400 } });
      if (geoRes.ok) {
        const geoData = await geoRes.json();
        const first = geoData?.results?.[0];
        if (first) {
          target = {
            name: first.country ? `${first.name} (${first.country})` : first.name,
            lat: first.latitude,
            lon: first.longitude,
          };
        }
      }
    } catch (_) {}
  }

  if (!target) {
    return `Không tìm thấy địa điểm "${rawLoc}" để xem thời tiết.`;
  }

  try {
    const wUrl = `https://api.open-meteo.com/v1/forecast?latitude=${target.lat}&longitude=${target.lon}&current=temperature_2m,apparent_temperature,relative_humidity_2m,weather_code,wind_speed_10m&daily=temperature_2m_max,temperature_2m_min,precipitation_probability_max&timezone=auto&forecast_days=1`;
    const res = await fetch(wUrl, { signal: AbortSignal.timeout(2500), cf: { cacheTtl: 600 } });
    if (!res.ok) return `Lỗi lấy dữ liệu thời tiết (HTTP ${res.status}).`;
    const data = await res.json();
    const cur = data.current || {};
    const day = data.daily || {};
    const desc = describeWeatherCode(cur.weather_code);
    const minT = day.temperature_2m_min?.[0];
    const maxT = day.temperature_2m_max?.[0];
    const rainP = day.precipitation_probability_max?.[0];
    return `Thời tiết ${target.name} (${formatVnDateTime(new Date())}): ${desc}, nhiệt độ hiện tại ${cur.temperature_2m}°C (cảm giác ${cur.apparent_temperature}°C), độ ẩm ${cur.relative_humidity_2m}%, gió ${cur.wind_speed_10m} km/h. Hôm nay thấp nhất ${minT}°C, cao nhất ${maxT}°C, khả năng mưa ${rainP}%.`;
  } catch (e) {
    return `Không thể kết nối máy chủ thời tiết lúc này (${e.message}).`;
  }
}

async function fetchWikiInLang(keyword, lang) {
  const searchUrl = `https://${lang}.wikipedia.org/w/api.php?action=query&list=search&srsearch=${encodeURIComponent(keyword)}&srlimit=2&format=json&utf8=1`;
  const sRes = await fetch(searchUrl, {
    headers: { "User-Agent": WIKI_UA },
    signal: AbortSignal.timeout(2500),
    cf: { cacheTtl: 1800 },
  });
  if (!sRes.ok) return null;
  const sData = await sRes.json();
  const hits = sData?.query?.search || [];
  for (const hit of hits) {
    const sumUrl = `https://${lang}.wikipedia.org/api/rest_v1/page/summary/${encodeURIComponent(hit.title)}`;
    const sumRes = await fetch(sumUrl, {
      headers: { "User-Agent": WIKI_UA },
      signal: AbortSignal.timeout(2500),
      cf: { cacheTtl: 1800 },
    });
    if (!sumRes.ok) continue;
    const sum = await sumRes.json();
    if (sum?.extract && sum.type !== "disambiguation") {
      let text = sum.extract.replace(/\s+/g, " ").trim();
      if (text.length > 450) {
        const cut = text.slice(0, 450);
        const lastDot = cut.lastIndexOf(".");
        text = lastDot > 200 ? cut.slice(0, lastDot + 1) : cut + "...";
      }
      return { title: sum.title || hit.title, extract: text, lang };
    }
  }
  return null;
}

async function fetchWikipedia(keyword, language) {
  const cleanKw = String(keyword || "").trim().slice(0, 120);
  if (!cleanKw) return "Thiếu từ khóa tra cứu Wikipedia.";
  const primary = language === "en" ? "en" : "vi";
  const secondary = primary === "vi" ? "en" : "vi";

  try {
    let result = await fetchWikiInLang(cleanKw, primary);
    if (!result) result = await fetchWikiInLang(cleanKw, secondary);
    if (result) {
      return `Wikipedia (${result.lang}) - ${result.title}: ${result.extract}`;
    }
  } catch (_) {}

  return `Không tìm thấy bài viết Wikipedia rõ ràng cho từ khóa "${cleanKw}".`;
}

async function fetchVietnamNews(topic, count) {
  const key = NEWS_FEEDS[topic] ? topic : "moi-nhat";
  const limit = Math.max(1, Math.min(5, Number(count) || 3));
  const feedUrl = NEWS_FEEDS[key];

  try {
    const res = await fetch(feedUrl, {
      headers: { "User-Agent": UA },
      signal: AbortSignal.timeout(2500),
      cf: { cacheTtl: 300 },
    });
    if (!res.ok) return `Không lấy được tin tức (HTTP ${res.status}).`;
    const xml = (await res.text()).slice(0, 28000);
    const items = [];
    RSS_ITEM_RE.lastIndex = 0;
    let m;
    while ((m = RSS_ITEM_RE.exec(xml)) !== null && items.length < limit) {
      const title = decodeHtml(m[1]);
      let desc = decodeHtml(m[2]);
      if (desc.length > 110) {
        const cut = desc.slice(0, 110);
        const lastSpace = cut.lastIndexOf(" ");
        desc = (lastSpace > 60 ? cut.slice(0, lastSpace) : cut) + "...";
      }
      const dateStr = formatVnDateTime(m[3].trim());
      items.push(`${items.length + 1}. [${dateStr}] ${title}${desc ? " - " + desc : ""}`);
    }
    if (items.length > 0) {
      return `Tin tức VnExpress (${key}), cập nhật ${formatVnDateTime(new Date())}. Khi đọc cho người dùng BẮT BUỘC nêu rõ ngày tháng xảy ra của từng tin:\n${items.join("\n")}`;
    }
  } catch (e) {
    return `Lỗi kết nối nguồn tin tức (${e.message}).`;
  }
  return "Hiện không đọc được danh sách tin tức mới nhất.";
}

function extractVideosFromText(rawText) {
  const sample = rawText.length > 220000 ? rawText.slice(0, 220000) : rawText;
  const list = [];
  const seen = new Set();
  YT_JSON_RE.lastIndex = 0;
  let match;
  while ((match = YT_JSON_RE.exec(sample)) !== null && list.length < 6) {
    if (seen.has(match[1])) continue;
    seen.add(match[1]);
    const title = match[2]
      .replace(/\\u0026/g, "&")
      .replace(/\\"/g, '"')
      .replace(/\\\\/g, "\\");
    list.push({ videoId: match[1], title });
  }
  return list;
}

async function fetchYouTubeRaw(searchQ, isLongStory, isEnglish, env, logFn) {
  const errors = [];
  const t0 = Date.now();

  const wrap = (name, fn) =>
    fn()
      .then((items) => {
        if (items && items.length > 0) return { source: name, items };
        errors.push(`${name}:empty`);
        throw new Error(`${name}:empty`);
      })
      .catch((e) => {
        if (!errors.some((x) => x.startsWith(name + ":"))) {
          errors.push(`${name}:${e.message || "err"}`);
        }
        throw e;
      });

  const lightTasks = [];

  if (env.YOUTUBE_API_KEY) {
    lightTasks.push(
      wrap("YT_API", async () => {
        const durationParam = isLongStory ? "&videoDuration=long" : "";
        const langParam = isEnglish ? "&relevanceLanguage=en" : "&relevanceLanguage=vi";
        const apiUrl = `https://www.googleapis.com/youtube/v3/search?part=snippet&type=video&videoEmbeddable=true&maxResults=6${durationParam}${langParam}&q=${encodeURIComponent(searchQ)}&key=${env.YOUTUBE_API_KEY}`;
        const res = await fetch(apiUrl, { signal: AbortSignal.timeout(2200), cf: { cacheTtl: 300 } });
        if (!res.ok) throw new Error(String(res.status));
        const data = await res.json();
        return (data.items || [])
          .filter((it) => it?.id?.videoId)
          .map((it) => ({ videoId: it.id.videoId, title: it.snippet.title }));
      })
    );
  }

  lightTasks.push(
    wrap("Piped", async () => {
      const pipedUrl = `https://api.piped.private.coffee/search?q=${encodeURIComponent(searchQ)}&filter=videos`;
      const res = await fetch(pipedUrl, {
        headers: { "User-Agent": UA },
        signal: AbortSignal.timeout(2200),
        cf: { cacheTtl: 300 },
      });
      if (!res.ok) throw new Error(String(res.status));
      const data = await res.json();
      const out = [];
      for (const it of data?.items || []) {
        const m = String(it?.url || "").match(/v=([a-zA-Z0-9_-]{11})/);
        if (m && it?.title) out.push({ videoId: m[1], title: String(it.title) });
        if (out.length >= 6) break;
      }
      return out;
    })
  );

  try {
    const winner = await Promise.any(lightTasks);
    logFn(`LOG: ⚡ Tìm thấy (${winner.source}, ${Date.now() - t0}ms)`);
    return winner.items;
  } catch (_) {}

  const fallbackTasks = [
    wrap("Innertube", async () => {
      const res = await fetch("https://www.youtube.com/youtubei/v1/search?prettyPrint=false", {
        method: "POST",
        headers: { "Content-Type": "application/json", "User-Agent": UA },
        body: JSON.stringify({
          context: {
            client: {
              clientName: "WEB",
              clientVersion: "2.20250501.00.00",
              hl: isEnglish ? "en" : "vi",
              gl: isEnglish ? "US" : "VN",
            },
          },
          query: searchQ,
        }),
        signal: AbortSignal.timeout(2500),
      });
      if (!res.ok) throw new Error(String(res.status));
      return extractVideosFromText(await res.text());
    }),
    wrap("Bing", async () => {
      const bingUrl = `https://www.bing.com/videos/search?q=${encodeURIComponent(searchQ + " youtube")}`;
      const res = await fetch(bingUrl, {
        headers: { "User-Agent": UA },
        signal: AbortSignal.timeout(2500),
        cf: { cacheTtl: 300 },
      });
      if (!res.ok) throw new Error(String(res.status));
      const html = await res.text();
      const seen = new Set();
      const out = [];
      BING_YT_RE.lastIndex = 0;
      let m;
      while ((m = BING_YT_RE.exec(html)) !== null && out.length < 5) {
        if (seen.has(m[1])) continue;
        seen.add(m[1]);
        out.push({ videoId: m[1], title: searchQ });
      }
      return out;
    }),
  ];

  try {
    const winner = await Promise.any(fallbackTasks);
    logFn(`LOG: ⚡ Tìm thấy (${winner.source}, ${Date.now() - t0}ms)`);
    return winner.items;
  } catch (_) {}

  if (errors.length > 0) {
    logFn("LOG: ⚠ Lỗi tìm YouTube: " + errors.join(" | "));
  }
  return [];
}

async function searchYouTubeCandidates(query, isLongStory, shuffle, language, env, logFn) {
  const cleanQ = isLongStory
    ? `${query} audio full`
    : query.replace(FILLER_RE, "").replace(/\s+/g, " ").trim() || query;
  const isEnglish = language === "en" || (language === "auto" && !VN_ACCENT_RE.test(cleanQ) && !isLongStory);

  let candidates = await fetchYouTubeRaw(cleanQ, isLongStory, isEnglish, env, logFn);

  if (!isLongStory && candidates.length > 1) {
    const qLower = cleanQ.toLowerCase();
    const scoreItem = (item) => {
      const t = item.title.toLowerCase();
      let score = 0;
      if (t.includes("official audio") || t.includes("lyric") || t.includes("official music video")) score += 3;
      if (t.includes("#shorts") || t.includes("reaction") || t.includes("1 hour")) score -= 5;
      if (t.includes("karaoke") && !qLower.includes("karaoke")) score -= 5;
      if (isEnglish && (t.includes("vietsub") || t.includes("cover")) && !qLower.includes("vietsub") && !qLower.includes("cover")) {
        score -= 4;
      }
      return score;
    };
    candidates.sort((a, b) => scoreItem(b) - scoreItem(a));
  }

  candidates = candidates.slice(0, 5);

  if (shuffle && candidates.length > 1) {
    const firstIdx = Math.floor(Math.random() * candidates.length);
    const [picked] = candidates.splice(firstIdx, 1);
    candidates.unshift(picked);
  }
  return candidates;
}

              audio: {
                mimeType: "audio/pcm;rate=16000",
                data: bytesToBase64(new Uint8Array(arrayBuffer)),
              },
            },
          })
        );
      }
    });

    geminiSocket.addEventListener("message", async (event) => {
      try {
        let text;
        if (typeof event.data === "string") {
          text = event.data;
        } else if (event.data instanceof ArrayBuffer) {
          text = TEXT_DECODER.decode(event.data);
        } else if (event.data instanceof Blob) {
          text = await event.data.text();
        } else {
          return;
        }

        const data = JSON.parse(text);

        if (data.toolCall?.functionCalls) {
          const functionResponses = [];

          for (const call of data.toolCall.functionCalls) {
            if (call.name === "save_temp_memory") {
              const note = call.args?.note || "";
              pushTempMemory(note);
              logFn("LOG: 🧠 Đã nhớ tạm: " + note);
              functionResponses.push({
                id: call.id,
                name: call.name,
                response: { result: `Đã lưu: "${note}".` },
              });
            } else if (call.name === "get_weather") {
              const loc = call.args?.location || "Hà Nội";
              logFn(`LOG: 🌤 Xem thời tiết: ${loc}`);
              const info = await fetchWeather(loc);
              logFn("LOG: " + info);
              functionResponses.push({
                id: call.id,
                name: call.name,
                response: { result: info },
              });
            } else if (call.name === "set_pomodoro") {
              const action = call.args?.action === "stop" ? "stop" : "start";
              const workMin = Math.max(1, Math.min(180, Number(call.args?.work_minutes) || 25));
              const rawBreak = call.args?.break_minutes;
              const breakMin =
                rawBreak === undefined || rawBreak === null || rawBreak === ""
                  ? 5
                  : Math.max(0, Math.min(60, Number(rawBreak) || 0));
              const label = String(call.args?.label || "Học tập").trim().slice(0, 40);
              safeSend(
                espSocket,
                JSON.stringify({
                  type: "POMODORO",
                  action,
                  workMinutes: workMin,
                  breakMinutes: breakMin,
                  label,
                })
              );
              const msg =
                action === "stop"
                  ? "Đã tắt đồng hồ Pomodoro."
                  : `Đã bật đồng hồ Pomodoro (${label}): tập trung ${workMin} phút${breakMin > 0 ? `, nghỉ ${breakMin} phút` : ""}.`;
              functionResponses.push({
                id: call.id,
                name: call.name,
                response: { result: msg },
              });
            } else if (call.name === "search_wikipedia") {
              const keyword = call.args?.keyword || "";
              const language = call.args?.language || "vi";
              logFn(`LOG: 📚 Tra Wikipedia [${language}]: ${keyword}`);
              const info = await fetchWikipedia(keyword, language);
              logFn("LOG: " + info.slice(0, 180) + (info.length > 180 ? "..." : ""));
              functionResponses.push({
                id: call.id,
                name: call.name,
                response: { result: info },
              });
            } else if (call.name === "get_news") {
              const topic = call.args?.topic || "moi-nhat";
              const count = Number(call.args?.count) || 3;
              logFn(`LOG: 📰 Đọc tin tức [${topic}] (${count} tin)`);
              const info = await fetchVietnamNews(topic, count);
              logFn("LOG: " + info);
              functionResponses.push({
                id: call.id,
                name: call.name,
                response: { result: info },
              });
            } else if (call.name === "play_youtube_music") {
              const query = call.args?.query || "Ed Sheeran Shape of You";
              const language = call.args?.language || "auto";
              const isRandom = Boolean(call.args?.is_random_mood);
              logFn(`LOG: 🎵 Tìm nhạc [${language}]: ${query}`);
              const candidates = await searchYouTubeCandidates(query, false, isRandom, language, env, logFn);
              if (candidates.length > 0) {
                pushTempMemory(`Đã phát: ${candidates[0].title}`);
                safeSend(
                  espSocket,
                  JSON.stringify({
                    type: "PLAY_YOUTUBE",
                    mode: "music",
                    candidates,
                    sleepMinutes: 0,
                  })
                );
                functionResponses.push({
                  id: call.id,
                  name: call.name,
                  response: {
                    result: `Đang phát: "${candidates[0].title}". Giới thiệu ngắn gọn.`,
                  },
                });
              } else {
                functionResponses.push({
                  id: call.id,
                  name: call.name,
                  response: { result: `Không tìm thấy bài hát cho ${query}.` },
                });
              }
            } else if (call.name === "play_audio_story") {
              const query = call.args?.query || "truyện ma Nguyễn Ngọc Ngạn";
              const sleepMinutes = Number(call.args?.sleep_minutes) || 45;
              const candidates = await searchYouTubeCandidates(query, true, false, "vi", env, logFn);
              if (candidates.length > 0) {
                pushTempMemory(`Đã phát truyện: ${candidates[0].title}`);
                safeSend(
                  espSocket,
                  JSON.stringify({
                    type: "PLAY_YOUTUBE",
                    mode: "story",
                    candidates,
                    sleepMinutes,
                  })
                );
                functionResponses.push({
                  id: call.id,
                  name: call.name,
                  response: {
                    result: `Đang phát truyện "${candidates[0].title}", tự tắt sau ${sleepMinutes} phút.`,
                  },
                });
              } else {
                functionResponses.push({
                  id: call.id,
                  name: call.name,
                  response: { result: `Không tìm thấy truyện ${query}.` },
                });
              }
            } else if (call.name === "stop_media") {
              const target = call.args?.target || "all";
              safeSend(espSocket, JSON.stringify({ type: "STOP_MEDIA", target }));
              functionResponses.push({
                id: call.id,
                name: call.name,
                response: { result: "Đã dừng phát." },
              });
            }
          }

          safeSend(geminiSocket, JSON.stringify({ toolResponse: { functionResponses } }));
          return;
        }

        const parts = data?.serverContent?.modelTurn?.parts;
        if (parts) {
          for (const part of parts) {
            if (part.inlineData?.mimeType?.startsWith("audio/pcm")) {
              safeSend(espSocket, base64ToBytes(part.inlineData.data).buffer);
            }
          }
        }
      } catch (err) {
        logFn("PARSE_ERR: " + err.message);
      }
    });

    espSocket.addEventListener("close", () => safeClose(geminiSocket, 1000, "Client closed"));
    geminiSocket.addEventListener("close", (e) => safeClose(espSocket, e.code, e.reason || "Gemini closed"));

    return new Response(null, { status: 101, webSocket: client });
  },
};
