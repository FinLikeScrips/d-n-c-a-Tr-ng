    let currentCandidates = [];
    let currentIdx = 0;

      if (ytPlayer && typeof ytPlayer.stopVideo === "function") {

      if (sleepTimer) {
        clearTimeout(sleepTimer);
        sleepTimer = null;
      }
    }

      threshVal.textContent = (threshSlider.value / 100).toFixed(2);

    function connectWebSocket() {
      if (userStopped) return;

      ws = new WebSocket(wsUrl);
      ws.binaryType = "arraybuffer";

      ws.onopen = function () {
        statusEl.textContent = "Đã kết nối! (Nhạc vẫn phát độc lập kể cả khi im lặng lâu)";

        stopBtn.disabled = false;

        silencePacketsLeft = 0;
        lastPacketSentTime = Date.now();
      };

      ws.onmessage = function (event) {
        if (typeof event.data === "string") {
          try {
            const cmd = JSON.parse(event.data);
            if (cmd.type === "PLAY_YOUTUBE") {
              if (sleepTimer) clearTimeout(sleepTimer);
              currentMode = cmd.mode || "music";
              currentCandidates = cmd.candidates || [{ videoId: cmd.videoId, title: cmd.title }];
              playCandidate(0);

              if (cmd.sleepMinutes > 0) {
                timerInfo.textContent = "(Tự tắt sau " + cmd.sleepMinutes + " phút)";

                  stopYouTube();
                  addLog("Đã tự động tắt truyện theo hẹn giờ.");

                timerInfo.textContent = "";
              }
              return;

              stopYouTube();
              addLog("⏹ Đã tắt nhạc/truyện.");
              return;
            }
          } catch (_) {}
          addLog(event.data);
          return;
        }

        if (!(event.data instanceof ArrayBuffer)) return;
        const int16 = new Int16Array(event.data);
        const float32 = new Float32Array(int16.length);
        for (let i = 0; i < int16.length; i++) float32[i] = int16[i] / 32768;

        src.buffer = buffer;
        src.connect(playCtx.destination);
        const startTime = Math.max(playCtx.currentTime, nextPlayTime);
        src.start(startTime);
        nextPlayTime = startTime + buffer.duration;

    startBtn.onclick = async function () {

        await recCtx.resume();
        await playCtx.resume();

        const rms = Math.sqrt(sum / float32.length);

          gateStatusEl.textContent = "Im lặng (Tiết kiệm quota)";
          gateStatusEl.style.color = "gray";

      processor.connect(recCtx.destination);

      if (playCtx && playCtx.state !== "closed") playCtx.close();
    };
  </script>
</body>
</html>`;

    const decoder = new TextDecoder();

    async function fetchVietnamGoldPrice() {
      const vnTime = new Date().toLocaleString("vi-VN", {
        timeZone: "Asia/Ho_Chi_Minh",
        hour12: false,
      });

      try {
        const res = await fetch("https://sjc.com.vn/xml/tygiavang.xml", {
          headers: { "User-Agent": "Mozilla/5.0" },
          signal: AbortSignal.timeout(4500),
          cf: { cacheTtl: 180 },
        });
        if (res.ok) {
          const xml = await res.text();
          const items = [];
          const regex = /<item\s+buy="([^"]+)"\s+sell="([^"]+)"\s+type="([^"]+)"/g;
          let m;
          while ((m = regex.exec(xml)) !== null) {
            const [, buy, sell, type] = m;
            if (
              type.includes("SJC 1L") ||
              type.includes("Nhẫn SJC 99,99") ||
              type.includes("Vàng miếng SJC")
            ) {
              items.push(`${type}: mua vào ${buy} triệu/lượng, bán ra ${sell} triệu/lượng`);
            }
            if (items.length >= 2) break;
          }
          if (items.length > 0) {
            return `Giờ Việt Nam: ${vnTime}. Giá vàng SJC cập nhật: ${items.join("; ")}.`;
          }
        }
      } catch (_) {}

      try {
        const res = await fetch("https://giavang.org/", {
          headers: { "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64)" },
          signal: AbortSignal.timeout(4500),
          cf: { cacheTtl: 180 },
        });
        if (res.ok) {
          const html = await res.text();
          const sjcMatch = html.match(/SJC[\s\S]{0,400}?(\d{2,3}\.\d{3})[\s\S]{0,150}?(\d{2,3}\.\d{3})/);
          if (sjcMatch) {
            return `Giờ Việt Nam: ${vnTime}. Giá vàng SJC tham khảo: mua vào ${sjcMatch[1]} triệu/lượng, bán ra ${sjcMatch[2]} triệu/lượng.`;
          }
        }
      } catch (_) {}

      return `Hiện tại là ${vnTime} giờ Việt Nam. Tạm thời nguồn bảng giá vàng trực tuyến đang bảo trì.`;
    }

    async function searchYouTubeCandidates(query, isLongStory = false, shuffle = false) {
      const searchQ = isLongStory ? `${query} audio full` : query;
      let candidates = [];

      if (env.YOUTUBE_API_KEY) {
        const durationParam = isLongStory ? "&videoDuration=long" : "";
        const apiUrl = `https://www.googleapis.com/youtube/v3/search?part=snippet&type=video&videoEmbeddable=true&maxResults=5${durationParam}&q=${encodeURIComponent(searchQ)}&key=${env.YOUTUBE_API_KEY}`;
        const res = await fetch(apiUrl);
        const data = await res.json();
        candidates = (data.items || []).map((it) => ({
          videoId: it.id.videoId,
          title: it.snippet.title,
        }));
      } else {
        const ytUrl = `https://www.youtube.com/results?search_query=${encodeURIComponent(searchQ)}`;
        const res = await fetch(ytUrl, {
          headers: {
            "User-Agent":
              "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36",
            "Accept-Language": "vi-VN,vi;q=0.9",
          },
        });
        const html = await res.text();
        const regex = /"videoRenderer":\{"videoId":"([a-zA-Z0-9_-]{11})".*?"title":\{"runs":\[\{"text":"([^"]+)"/g;
        const seen = new Set();
        let match;
        while ((match = regex.exec(html)) !== null && candidates.length < 5) {
          if (seen.has(match[1])) continue;
          seen.add(match[1]);
          const title = match[2].replace(/\\u0026/g, "&").replace(/\\"/g, '"');
          candidates.push({ videoId: match[1], title });
        }
      }

      if (shuffle && candidates.length > 1) {
        const firstIdx = Math.floor(Math.random() * candidates.length);
        const [picked] = candidates.splice(firstIdx, 1);
        candidates.unshift(picked);
      }
      return candidates;
    }

    espSocket.addEventListener("message", async (event) => {
      let arrayBuffer = null;
      if (event.data instanceof ArrayBuffer) {
        arrayBuffer = event.data;
      } else if (event.data instanceof Blob) {
        arrayBuffer = await event.data.arrayBuffer();
      }

      if (arrayBuffer && arrayBuffer.byteLength > 0) {
        const bytes = new Uint8Array(arrayBuffer);
        let binary = "";
        for (let i = 0; i < bytes.byteLength; i++) {
          binary += String.fromCharCode(bytes[i]);
        }
        geminiSocket.send(
          JSON.stringify({
            realtimeInput: {
              audio: {
                mimeType: "audio/pcm;rate=16000",
                data: btoa(binary),
              },
            },
          })
        );
      }
    });

    geminiSocket.addEventListener("message", async (event) => {
      try {
        let text;
        if (event.data instanceof Blob) {
          text = await event.data.text();
        } else if (event.data instanceof ArrayBuffer) {
          text = decoder.decode(event.data);
        } else {
          text = event.data;
        }

        const data = JSON.parse(text);

        if (data.toolCall?.functionCalls) {
          const functionResponses = [];

          for (const call of data.toolCall.functionCalls) {
            if (call.name === "get_gold_price") {
              const info = await fetchVietnamGoldPrice();
              espSocket.send("LOG: " + info);
              functionResponses.push({
                id: call.id,
                name: call.name,
                response: { result: info },
              });
            } else if (call.name === "play_youtube_music") {
              const query = call.args?.query || "Lofi chill Việt Nam lyric video";
              const isRandom = Boolean(call.args?.is_random_mood);
              const candidates = await searchYouTubeCandidates(query, false, isRandom);
              if (candidates.length > 0) {
                espSocket.send(
                  JSON.stringify({
                    type: "PLAY_YOUTUBE",
                    mode: "music",
                    candidates,
                    sleepMinutes: 0,
                  })
                );
                functionResponses.push({
                  id: call.id,
                  name: call.name,
                  response: {
                    result: `Đang phát trên YouTube: "${candidates[0].title}". Giới thiệu ngắn tên bài này cho người dùng.`,
                  },
                });
              } else {
                functionResponses.push({
                  id: call.id,
                  name: call.name,
                  response: { result: `Không tìm thấy bài hát phù hợp cho ${query}.` },
                });
              }
            } else if (call.name === "play_audio_story") {
              const query = call.args?.query || "truyện ma Nguyễn Ngọc Ngạn";
              const sleepMinutes = Number(call.args?.sleep_minutes) || 45;
              const candidates = await searchYouTubeCandidates(query, true, false);
              if (candidates.length > 0) {
                espSocket.send(
                  JSON.stringify({
                    type: "PLAY_YOUTUBE",
                    mode: "story",
                    candidates,
                    sleepMinutes,
                  })
                );
                functionResponses.push({
                  id: call.id,
                  name: call.name,
                  response: {
                    result: `Đang phát truyện "${candidates[0].title}" và hẹn giờ tự tắt sau ${sleepMinutes} phút.`,
                  },
                });
              } else {
                functionResponses.push({
                  id: call.id,
                  name: call.name,
                  response: { result: `Không tìm thấy truyện ${query}.` },
                });
              }
            } else if (call.name === "stop_media") {
              espSocket.send(JSON.stringify({ type: "STOP_MEDIA" }));
              functionResponses.push({
                id: call.id,
                name: call.name,
                response: { result: "Đã dừng phát." },
              });
            }
          }

          geminiSocket.send(JSON.stringify({ toolResponse: { functionResponses } }));
          return;
        }

        const parts = data?.serverContent?.modelTurn?.parts;
        if (parts) {
          for (const part of parts) {
            if (part.inlineData?.mimeType?.startsWith("audio/pcm")) {
              const binary = atob(part.inlineData.data);
              const pcmBytes = new Uint8Array(binary.length);
              for (let i = 0; i < binary.length; i++) {
                pcmBytes[i] = binary.charCodeAt(i);
              }
              espSocket.send(pcmBytes.buffer);
            }
          }
        }
      } catch (err) {
        espSocket.send("PARSE_ERR: " + err.message);
      }
    });

    espSocket.addEventListener("close", () => geminiSocket.close());
    geminiSocket.addEventListener("close", (e) => {
      espSocket.close(e.code || 1000, e.reason || "Gemini closed");
    });

    return new Response(null, { status: 101, webSocket: client });
  },
};
